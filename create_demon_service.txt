sudo vim /etc/systemd/system/faceRegistration_service.service

[Unit]
Description=Face Registration Service
After=network.target

[Service]
User=root
Group=root
WorkingDirectory=/devfacerec_devwebapp/
Environment="PATH=/devfacerec_devwebapp/env/bin"

# The magic happens here: We call gunicorn, but tell it to use Uvicorn workers
ExecStart=/devfacerec_devwebapp/env/bin/gunicorn app:app \
          --workers 3 \
          --worker-class uvicorn.workers.UvicornWorker \
          --timeout 1200 \
          --bind 127.0.0.1:8000

Restart=always

# I updated the log file names to reflect gunicorn, but you can keep them as uvicorn.log if you prefer!
StandardOutput=append:/devfacerec_devwebapp/gunicorn.log
StandardError=append:/devfacerec_devwebapp/gunicorn.log

[Install]
WantedBy=multi-user.target


sudo systemctl daemon-reload
sudo systemctl enable faceRegistration_service
sudo systemctl start faceRegistration_service
sudo systemctl status faceRegistration_service
sudo systemctl restart faceRegistration_service
sudo systemctl stop faceRegistration_service

cp -r /devfacerec_devwebapp/dev/* /devfacerec_devwebapp/

tail -f /devfacerec_devwebapp/gunicorn.log
tail -f /devfacerec_devwebapp/gunicorn.log


*/3 * * * * /devfacerec_devwebapp/env/bin/gunicorn app:app -k uvicorn.workers.UvicornWorker --workers 3 --timeout 1200 --bind 127.0.0.1:8000 >> /devfacerec_devwebapp/gunicorn.log 2>&1
